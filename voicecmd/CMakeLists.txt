# Build voicecmd against an existing whisper.cpp tree (host or QNX cross build).
# cmake -S voicecmd -B build -DWHISPER_DIR=/path/to/whisper.cpp [-DCMAKE_TOOLCHAIN_FILE=...]
cmake_minimum_required(VERSION 3.16)
project(voicecmd C CXX)
set(CMAKE_CXX_STANDARD 17)
set(WHISPER_DIR "" CACHE PATH "whisper.cpp source tree (already built; its libs are in WHISPER_BUILD)")
set(WHISPER_BUILD "" CACHE PATH "whisper.cpp build dir containing src/libwhisper and ggml/src libs")
option(HAVE_QNX_SND "Enable the io-snd capture backend (needs libasound headers)" OFF)
add_library(grammar STATIC src/grammar.cpp)
target_include_directories(grammar PUBLIC src)
add_executable(voicecmd src/voicecmd.cpp src/audio_source.c)
target_include_directories(voicecmd PRIVATE src ${WHISPER_DIR}/include ${WHISPER_DIR}/ggml/include)
target_link_directories(voicecmd PRIVATE ${WHISPER_BUILD}/bin)
target_link_libraries(voicecmd grammar whisper ggml ggml-base ggml-cpu)
if(HAVE_QNX_SND)
  target_compile_definitions(voicecmd PRIVATE HAVE_QNX_SND)
  target_link_libraries(voicecmd asound)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "QNX")
  target_link_libraries(voicecmd socket)
endif()
add_executable(test_grammar tests/test_grammar.cpp)
target_link_libraries(test_grammar grammar)
